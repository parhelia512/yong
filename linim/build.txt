// vim: set ft=javascript:
const SRCS=[
	'main.c','im.c','bihua.c','common.c','s2t.c','nl.c',
	'english.c','dict.c','keyboard.c',
	'aes.c','book.c','keytool.c',
	'keydesc.c','ui.c','select.c','layout.c','translate.c',
	'ybus.c','ybus-xim.c','ybus-lcall.c','lcall.c','ybus-ibus.c',
	'menu.c','history.c','replace.c','ybus-wayland.c',
	'app.c','async.c','ui-draw.c','speed.c','color-schema.c',
	'app-indicator.c',
];

const DIRS=['l64'];

feature(['x11','ibus']);

env("VPATH+=$(TOPDIR)/linim");
env("VPATH+=$(TOPDIR)/linim/wayland");

env("CFLAGS+=$(LLIB_CFLAGS)");
env("LDFLAGS+=$(LLIB_LDFLAGS)");
env("LIBS+=$(LLIB_LDADD)");

env("CFLAGS+=-I$(TOPDIR)/linim");
env("CFLAGS+=-I$(TOPDIR)/linim/wayland");
env("CFLAGS+=-DCFG_XIM_YBUS");
env("CFLAGS+=-DCFG_XIM_IBUS `pkg-config --cflags ibus-1.0`");
env("CFLAGS+=-DL_CALL_GLIB_SERVER -DL_CALL_GLIB_CLIENT");
env("CFLAGS+=-Wno-deprecated-declarations");
if(support("x11"))
	env("CFLAGS+=-I$(TOPDIR)/linim/IMdkit");
env("CFLAGS+=-I$(TOPDIR)/linim/wui");

env("LDFLAGS+=-Wl,-rpath,'$ORIGIN'");
env("LIBS+=-ll -ldl -lm -lxkbcommon");

env('CFLAGS+=-DUSE_WUI');

function run(target){
	cd(target);
	include("../../rules.txt");
	env("CFLAGS+=`pkg-config --cflags glib-2.0 gio-2.0 pango pangocairo gdk-pixbuf-2.0` -Wno-unused-const-variable");
	env("LIBS+=`pkg-config --libs glib-2.0 gio-2.0 pango pangocairo gdk-pixbuf-2.0`");
	var OBJS=wildcard(SRCS,function(input){
		return input.replace(/\.c$/,'.o');
	});
	begin();
	cc(SRCS,OBJS);
	end(function(){
		env("LDFLAGS+=-L$(TOPDIR)/linim/IMdkit/$(BUILD)");
		if(support('x11'))
			env('LIBS+=-lX11 -lXimd');
		ld(OBJS,'$(DESTDIR)/yong');
	});
}

function bihua(){
	var SRCS=['bihua.c'];
	env("BUILD?=l64");
	include("../rules.txt");
	env("CFLAGS+=-I$(TOPDIR)/include $(LLIB_CFLAGS) -Wall -DTOOLS_BIHUA");
	env("LDFLAGS+=-Wl,-rpath='$ORIGIN'");
	cr(SRCS,'l64/bihua','$(CC) $(CFLAGS) $(LDFLAGS) $^ -o $@ $(LLIB_LDADD)');
}

if(target=="clean"){
	rmdir(DIRS,'*');
	build(["IMdkit","wui"],undefined,'clean');
} else if(!target || target=="all"){
	build(["IMdkit","wui"]);
	build(undefined,undefined,DIRS);
} else if(target=='l64') {
	if(recursive){
		run(target);
	}else{
		build(["IMdkit","wui"],undefined,'l64');
		build(undefined,undefined,'l64');
	}
}else if(target=="bihua") {
	bihua();
} else if(target=="test") {
	exec("install l64/yong ../install/yong/l64/yong-wui");
} else {
	run(target);
}
